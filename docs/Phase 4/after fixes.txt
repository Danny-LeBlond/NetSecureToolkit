VALID_USERNAME = "admin"
PASSWORD = "password"

def login():
    # Limit login attempts to 3 to prevent brute force guessing of credentials
    attempts = 0

    while attempts < 3:  # Only allow 3 tries before stopping
        user = input("Enter username: ")
        pwd = input("Enter password: ")

        # Compare user input to stored password instead of trusting input from user
        # This is safer because it checks the input against known password before letting them in
        if user == VALID_USERNAME and pwd == PASSWORD:
            print("Login successful")
            return True
        else:
            print("Login failed")
            attempts += 1

    print("Exiting.")
    return False  

def run_command():
    # Only allow safe commands instead of letting user run anything  
    allowed_commands = ["whoami", "dir", "hostname", "date /t", "time /t", "echo hello"]
    attempts = 0  # Track invalid command attempts

    while attempts < 5:  # Allow up to 5  tries
        
        user_input = input("Enter a system command: ")

        # Check if command is in allowed list
        if user_input in allowed_commands:
            print("Executing command...")
            os.system(user_input)
            return
        else:
            print("Command not allowed. Allowed commands: whoami, dir, hostname, date /t, time /t, echo hello")
            attempts += 1  # Count bad attempts

    # Exit after too many invalid attempts
    print("Exiting.")

        print("Invalid input")

        print("Input too long")

        raw_bytes = data.encode('latin1')

def unsafe_eval():

    # Only allow digits, spaces, and basic math operators
    allowed_chars = set("0123456789+-*/%() .")

    expr = input("Enter a Python expression: ")

    # Reject empty input so it does not try to evaluate blank input which can cause errors
    if not expr.strip():
        print("Invalid input")
        return

    # Check every character is allowed so the user cannot include letters or code keywords
    # This helps block dangerous input and limits eval to only simple math expressions
    for ch in expr:
        if ch not in allowed_chars:
            print("Invalid")
            return

    try:
        # This is safer because only simple math characters are allowed, so eval cannot run full Python code
        result = eval(expr)
        print("Result:", result)
    except:
        # This is safer because bad input is handled without crashing the program
        print("Invalid expression")

    filename = input("Enter filename only: ")
    content = input("Enter content: ")

    # Reject file paths and only allow a simple file name
    if "\" in filename or "/" in filename or "." in filename:
        print("Only the name")
        return

    # Build a fixed safe base path (Desktop) so the user cannot choose arbitrary locations
    # This is safer because it prevents writing files anywhere
    desktop_path = os.path.join(os.path.expanduser("~"), "Desktop")

    # Create the final path using only the validated filename and a forced .txt extension
    # This is safer because it blocks other file types like .exe, .bat, or .py, ect
    full_path = os.path.join(desktop_path, filename + ".txt")

    with open(full_path, 'w') as f:
        f.write(content)

    print("File written.")
