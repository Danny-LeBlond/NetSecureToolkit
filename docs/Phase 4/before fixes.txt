import os
import pickle
import socket

def login():

        user = input("Enter username: ")
        pwd = input("Enter password: ")

            print("Login successful")
        else:
            print("Login failed")
    except:
        pass  

def run_command():

    # Dangerous examples:
    # Linux/macOS: rm -rf ~/testdir
    # Windows: rmdir /s /q C:\\Users\\Public\\testdir

    print("Executing command...")
    os.system(user_input)


def deserialize_data():
    data = input("Enter serialized data: ")
    raw_bytes = data.encode('latin1')
    obj = pickle.loads(raw_bytes)  
    print("Deserialized object:", obj)


def unsafe_eval():
    expr = input("Enter a Python expression: ")
    result = eval(expr)  
    print("Result:", result)


def write_file():
    filename = input("Enter filename: ")
    content = input("Enter content: ")

    with open(filename, 'w') as f:
        f.write(content)

    print("File written.")
